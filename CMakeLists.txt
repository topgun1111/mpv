cmake_minimum_required(VERSION 3.21)
project(TakipliOynatNative VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(native_camera INTERFACE)
target_include_directories(native_camera INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(BUILD_TESTING)
 add_executable(native_camera_test tests/test_track_math.cpp)
 target_link_libraries(native_camera_test PRIVATE native_camera)
 add_test(NAME camera_math COMMAND native_camera_test)
 add_executable(native_target_tracker_test tests/test_target_tracker.cpp)
 target_link_libraries(native_target_tracker_test PRIVATE native_camera)
 add_test(NAME target_tracker COMMAND native_target_tracker_test)
endif()
option(BUILD_NATIVE_GUI "Build Qt6/OpenCV desktop application" ON)
if(BUILD_NATIVE_GUI)
 find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
 find_package(OpenCV 4.5 REQUIRED COMPONENTS core imgproc imgcodecs videoio objdetect dnn)
 add_executable(TakipliOynatNative WIN32 src/main.cpp src/pipeline.cpp src/pipeline.hpp)
 target_link_libraries(TakipliOynatNative PRIVATE Qt6::Widgets native_camera ${OpenCV_LIBS})
 target_include_directories(TakipliOynatNative PRIVATE ${OpenCV_INCLUDE_DIRS})
endif()
